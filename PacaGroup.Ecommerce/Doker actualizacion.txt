# 1. Construir la imagen con la última versión del código
docker build -t pacagroup-ecommerce-api .

docker stop pacagroup-ecommerce-api 2>$null
docker rm pacagroup-ecommerce-api 2>$null

# 3. Crear y correr un nuevo contenedor
docker run -d -p 8080:8080 --name pacagroup-ecommerce-api pacagroup-ecommerce-api
